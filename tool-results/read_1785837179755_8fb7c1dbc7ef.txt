   241→function UserDialog({
   242→  user,
   243→  onClose,
   244→  onSaved,
   245→}: {
   246→  user: DbUser | null
   247→  onClose: () => void
   248→  onSaved: () => void
   249→}) {
   250→  const isEdit = !!user
   251→  const [name, setName] = useState(user?.name ?? '')
   252→  const [email, setEmail] = useState(user?.email ?? '')
   253→  const [password, setPassword] = useState('')
   254→  const [role, setRole] = useState<Role>(user?.role ?? 'EMPLOYEE')
   255→  const [isActive, setIsActive] = useState(user?.isActive ?? true)
   256→  const [customFields, setCustomFields] = useState<Record<string, string>>(user?.customFields || {})
   257→  const [saving, setSaving] = useState(false)
   258→
   259→  async function save() {
   260→    if (!name.trim() || !email.trim()) {
   261→      toast.error('Nombre y email son obligatorios')
   262→      return
   263→    }
   264→    setSaving(true)
   265→    try {
   266→      const body: Record<string, unknown> = {
   267→        name: name.trim(),
   268→        email: email.trim(),
   269→        role,
   270→        isActive,
   271→        customFields,
   272→      }
   273→      if (password) body.password = password
   274→      if (isEdit && user) {
   275→        await put(`/api/users/${user.id}`, body)
   276→        toast.success('Usuario actualizado')
   277→      } else {
   278→        if (!password) {
   279→          toast.error('La contraseña es obligatoria al crear')
   280→          setSaving(false)
   281→          return
   282→        }
   283→        await post('/api/users', body)
   284→        toast.success('Usuario creado')
   285→      }
   286→      onSaved()
   287→    } catch {
   288→      toast.error('No se pudo guardar')
   289→    } finally {
   290→      setSaving(false)
   291→    }
   292→  }
   293→
   294→  return (
   295→    <ModalShell
   296→      open
   297→      onClose={onClose}
   298→      title={isEdit ? 'Editar empleado' : 'Nuevo empleado'}
   299→      size="lg"
   300→      footer={
   301→        <>
   302→          <button className="btn-ghost text-sm" onClick={onClose}>
   303→            Cancelar
   304→          </button>
   305→          <button className="btn-sage text-sm" onClick={save} disabled={saving}>
   306→            <Save className="w-4 h-4" /> {saving ? 'Guardando…' : 'Guardar'}
   307→          </button>
   308→        </>
   309→      }
   310→    >
   311→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   312→        <div>
   313→          <label className="text-sm font-medium block mb-1.5">Nombre *</label>
   314→          <input className="input-wellness" value={name} onChange={(e) => setName(e.target.value)} />
   315→        </div>
   316→        <div>
   317→          <label className="text-sm font-medium block mb-1.5">Email *</label>
   318→          <input className="input-wellness" type="email" value={email} onChange={(e) => setEmail(e.target.value)} />
   319→        </div>
   320→        <div>
   321→          <label className="text-sm font-medium block mb-1.5">Contraseña {isEdit && '(dejar vacío para mantener)'}</label>
   322→          <input
   323→            className="input-wellness"
   324→            type="password"
   325→            value={password}
   326→            onChange={(e) => setPassword(e.target.value)}
   327→            placeholder={isEdit ? '••••••' : 'Contraseña inicial'}
   328→          />
   329→        </div>
   330→        <div>
   331→          <label className="text-sm font-medium block mb-1.5">Rol</label>
   332→          <select className="input-wellness" value={role} onChange={(e) => setRole(e.target.value as Role)}>
   333→            <option value="ADMIN">ADMIN</option>
   334→            <option value="EMPLOYEE">EMPLOYEE</option>
   335→            <option value="COCINERO">COCINERO</option>
   336→            <option value="ANOTADOR">ANOTADOR</option>
   337→          </select>
   338→        </div>
   339→        <div className="sm:col-span-2">
   340→          <label className="inline-flex items-center gap-2 cursor-pointer">
   341→            <input
   342→              type="checkbox"
   343→              checked={isActive}
   344→              onChange={(e) => setIsActive(e.target.checked)}
   345→              className="w-4 h-4 accent-[color:var(--sage)]"
   346→            />
   347→            <span className="text-sm">Usuario activo</span>
   348→          </label>
   349→        </div>
   350→        <div className="sm:col-span-2">
   351→          <KVEditor value={customFields} onChange={setCustomFields} />
   352→        </div>
   353→      </div>
   354→    </ModalShell>
   355→  )
   356→}

   526→function ShiftDialog({
   527→  shift,
   528→  onClose,
   529→  onSaved,
   530→}: {
   531→  shift: Shift | null
   532→  onClose: () => void
   533→  onSaved: () => void
   534→}) {
   535→  const isEdit = !!shift
   536→  const [name, setName] = useState(shift?.name ?? '')
   537→  const [startTime, setStartTime] = useState(shift?.startTime ?? '07:00')
   538→  const [endTime, setEndTime] = useState(shift?.endTime ?? '14:00')
   539→  const [days, setDays] = useState<number[]>(parseDays(shift?.daysOfWeek).length ? parseDays(shift?.daysOfWeek) : [1, 2, 3, 4, 5])
   540→  const [openingProtocol, setOpeningProtocol] = useState((shift?.openingProtocol || []).join('\n'))
   541→  const [closingProtocol, setClosingProtocol] = useState((shift?.closingProtocol || []).join('\n'))
   542→  const [saving, setSaving] = useState(false)
   543→
   544→  function toggleDay(d: number) {
   545→    setDays((prev) => (prev.includes(d) ? prev.filter((x) => x !== d) : [...prev, d].sort()))
   546→  }
   547→
   548→  async function save() {
   549→    if (!name.trim()) {
   550→      toast.error('El nombre es obligatorio')
   551→      return
   552→    }
   553→    setSaving(true)
   554→    try {
   555→      const body = {
   556→        name: name.trim(),
   557→        startTime,
   558→        endTime,
   559→        daysOfWeek: days.join(','),
   560→        openingProtocol: openingProtocol.split('\n').map((s) => s.trim()).filter(Boolean),
   561→        closingProtocol: closingProtocol.split('\n').map((s) => s.trim()).filter(Boolean),
   562→      }
   563→      if (isEdit && shift) {
   564→        await put(`/api/shifts/${shift.id}`, body)
   565→        toast.success('Turno actualizado')
   566→      } else {
   567→        await post('/api/shifts', body)
   568→        toast.success('Turno creado')
   569→      }
   570→      onSaved()
   571→    } catch {

   630→            onChange={(e) => setOpeningProtocol(e.target.value)}

   648→function DeudasTab() {
   649→  const [items, setItems] = useState<ShiftDebt[]>([])
   650→  const [loading, setLoading] = useState(true)
   651→  const [filter, setFilter] = useState<'all' | 'pending' | 'paid'>('pending')
   652→
   653→  async function load() {
   654→    setLoading(true)
   655→    try {
   656→      const data = await get<ShiftDebt[]>('/api/shift-debts')
   657→      setItems(data)
   658→    } catch {
   659→      toast.error('No se pudieron cargar las deudas')
   660→    } finally {
   661→      setLoading(false)
   662→    }
   663→  }
   664→  useEffect(() => {
   665→    load()
   666→  }, [])
   667→
   668→  const filtered = useMemo(() => {
   669→    if (filter === 'pending') return items.filter((d) => !d.isPaid)
   670→    if (filter === 'paid') return items.filter((d) => d.isPaid)
   671→    return items
   672→  }, [items, filter])
   673→
   674→  return (
   675→    <div>
   676→      <Toolbar>
   677→        <div className="inline-flex bg-muted/60 rounded-full p-1">
   678→          {([
   679→            ['pending', 'Pendientes'],
   680→            ['paid', 'Pagadas'],
   681→            ['all', 'Todas'],
   682→          ] as const).map(([id, label]) => (
   683→            <button
   684→              key={id}
   685→              onClick={() => setFilter(id)}
   686→              className={`px-3 py-1.5 text-xs font-medium rounded-full transition-colors ${
   687→                filter === id ? 'bg-card text-sage shadow-sm' : 'text-muted-foreground hover:text-foreground'
   688→              }`}
   689→            >
   690→              {label}
   691→            </button>
   692→          ))}
   693→        </div>
   694→      </Toolbar>
   695→
   696→      <Card className="p-0 overflow-hidden">
   697→        {loading ? (
   698→          <LoadingBlock label="Cargando deudas…" />
   699→        ) : filtered.length === 0 ? (
   700→          <EmptyState icon={<Coins className="w-6 h-6" />} title="Sin deudas" description="No hay deudas en este filtro." />
   701→        ) : (
   702→          <div className="overflow-x-auto custom-scroll">
   703→            <table className="w-full text-sm">
   704→              <thead className="bg-muted/40 text-muted-foreground text-xs uppercase tracking-wide">
   705→                <tr>
   706→                  <th className="text-left font-medium px-4 py-3">Empleado</th>
   707→                  <th className="text-left font-medium px-4 py-3">Motivo</th>
   708→                  <th className="text-left font-medium px-4 py-3">Estado</th>
   709→                  <th className="text-left font-medium px-4 py-3">Fecha</th>
   710→                  <th className="text-right font-medium px-4 py-3">Acciones</th>
   711→                </tr>
   712→              </thead>
   713→              <tbody className="divide-y divide-border">
   714→                {filtered.map((d) => (
   715→                  <tr key={d.id} className="hover:bg-accent/40">
   716→                    <td className="px-4 py-3 font-medium">{d.user?.name || '—'}</td>
   717→                    <td className="px-4 py-3 text-muted-foreground">{d.reason}</td>
   718→                    <td className="px-4 py-3">
   719→                      {d.isPaid ? (
   720→                        <Badge variant="sage"><CheckCircle2 className="w-3 h-3" /> Pagada</Badge>
   721→                      ) : (
   722→                        <Badge variant="warn"><XCircle className="w-3 h-3" /> Pendiente</Badge>
   723→                      )}
   724→                    </td>
   725→                    <td className="px-4 py-3 text-muted-foreground">{formatDate(d.createdAt)}</td>
   726→                    <td className="px-4 py-3 text-right">
   727→                      {!d.isPaid && (
   728→                        <button
   729→                          className="btn-sage text-xs px-3 py-1.5"
   730→                          onClick={async () => {
   731→                            try {
   732→                              await put(`/api/shift-debts/${d.id}/pay`)
   733→                              toast.success('Deuda marcada como pagada')
   734→                              load()
   735→                            } catch {
   736→                              toast.error('No se pudo actualizar')
   737→                            }
   738→                          }}
   739→                        >
   740→                          <CheckCircle2 className="w-3.5 h-3.5" /> Marcar pagada
   741→                        </button>
   742→                      )}
   743→                    </td>
   744→                  </tr>
   745→                ))}
   746→              </tbody>
   747→            </table>
   748→          </div>
   749→        )}
   750→      </Card>
   751→    </div>
   752→  )
   753→}
