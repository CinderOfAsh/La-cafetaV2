    42→    42→interface SandboxComanda {
    43→    43→  id: string
    44→    44→  name: string
    45→    45→  price: number
    46→    46→  priority: number
    47→    47→  payment: PaymentMethod

    50→    50→
    51→    51→export function SandboxPizarra({ products }: { products: Product[] }) {
    52→    52→  const [order, setOrder] = useState<string[]>(() => products.map((p) => p.id))
    53→    53→  const [search, setSearch] = useState('')
    54→    54→  const [activeTag, setActiveTag] = useState('todos')
    55→    55→  const [paymentModal, setPaymentModal] = useState<Product | null>(null)
    56→    56→  const [protocolModal, setProtocolModal] = useState<Protocol | null>(null)
    57→    57→  const [comandas, setComandas] = useState<SandboxComanda[]>([])
    58→    58→  const [protocols, setProtocols] = useState<Protocol[]>([])
    59→    59→
    60→    60→  useEffect(() => {
    61→    61→    setOrder(products.map((p) => p.id))
    62→    62→  }, [products])
    63→    63→
    64→    64→  useEffect(() => {
    65→    65→    get<Protocol[]>('/api/protocols').then(setProtocols).catch(() => {})
    66→    66→  }, [])
    67→    67→
    68→    68→  const tags = useMemo(() => {
    69→    69→    const s = new Set<string>()
    70→    70→    for (const p of products) for (const t of p.tags || []) s.add(t)
    71→    71→    return ['todos', ...Array.from(s).sort()]
    72→    72→  }, [products])
    73→    73→
    74→    74→  const filtered = useMemo(() => {
    75→    75→    const q = search.trim().toLowerCase()
    76→    76→    return order
    77→    77→      .map((id) => products.find((p) => p.id === id))
    78→    78→      .filter(Boolean)
    79→    79→      .filter((p) => (activeTag === 'todos' ? true : (p!.tags || []).includes(activeTag)))
    80→    80→      .filter((p) => (q ? p!.name.toLowerCase().includes(q) : true)) as Product[]
    81→    81→  }, [order, products, search, activeTag])
    82→    82→
    83→    83→  const sensors = useSensors(
    84→    84→    useSensor(PointerSensor, { activationConstraint: { distance: 6 } }),
    85→    85→    useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates })
    86→    86→  )
    87→    87→
    88→    88→  function onDragEnd(e: DragEndEvent) {
    89→    89→    const { active, over } = e
    90→    90→    if (!over || active.id === over.id) return
    91→    91→    setOrder((prev) => {
    92→    92→      const oldI = prev.indexOf(active.id as string)
    93→    93→      const newI = prev.indexOf(over.id as string)
    94→    94→      return arrayMove(prev, oldI, newI)
    95→    95→    })
    96→    96→  }
    97→    97→
    98→    98→  function addSale(product: Product, payment: PaymentMethod) {
    99→    99→    const id = `local-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`
   100→   100→    setComandas((prev) => [
   101→   101→      ...prev,
   102→   102→      {
   103→   103→        id,
   104→   104→        name: product.name,
   105→   105→        price: product.price,
   106→   106→        priority: prev.filter((c) => c.status === 'PENDING').length,
   107→   107→        payment,
   108→   108→        status: 'PENDING',
   109→   109→      },
   110→   110→    ])
   111→   111→    setPaymentModal(null)
   112→   112→    toast.success(`Venta simulada: ${product.name} (${payment === 'cash' ? 'efectivo' : 'tarjeta'})`)
   113→   113→  }
   114→   114→
   115→   115→  function move(id: string, dir: -1 | 1) {
   116→   116→    setComandas((prev) => {
   117→   117→      const pending = prev.filter((c) => c.status === 'PENDING')
   118→   118→      const idx = pending.findIndex((c) => c.id === id)
   119→   119→      if (idx === -1) return prev
   120→   120→      const newIdx = idx + dir
   121→   121→      if (newIdx < 0 || newIdx >= pending.length) return prev
   122→   122→      const reordered = [...pending]
   123→   123→      ;[reordered[idx], reordered[newIdx]] = [reordered[newIdx], reordered[idx]]
   124→   124→      // reassign priorities
   125→   125→      reordered.forEach((c, i) => (c.priority = i))
   126→   126→      // merge back: keep DELIVERED items first, then pending by priority
   127→   127→      const delivered = prev.filter((c) => c.status === 'DELIVERED')
   128→   128→      return [...delivered, ...reordered]
   129→   129→    })
   130→   130→  }
   131→   131→
   132→   132→  function markDone(id: string) {
   133→   133→    setComandas((prev) => {
   134→   134→      const next = prev.map((c) => (c.id === id ? { ...c, status: 'DELIVERED' as const } : c))
   135→   135→      // reassign priorities among pending
   136→   136→      let p = 0
   137→   137→      return next.map((c) => {
   138→   138→        if (c.status === 'PENDING') return { ...c, priority: p++ }
   139→   139→        return c
   140→   140→      })
   141→   141→    })
   142→   142→  }
   143→   143→
   144→   144→  function remove(id: string) {
   145→   145→    setComandas((prev) => {
   146→   146→      const next = prev.filter((c) => c.id !== id)
   147→   147→      let p = 0
   148→   148→      return next.map((c) => {
   149→   149→        if (c.status === 'PENDING') return { ...c, priority: p++ }
   150→   150→        return c
   151→   151→      })
   152→   152→    })
   153→   153→  }
   154→   154→
   155→   155→  const pending = comandas.filter((c) => c.status === 'PENDING')
   156→   156→  const delivered = comandas.filter((c) => c.status === 'DELIVERED')
   157→   157→  const total = delivered.reduce((s, c) => s + c.price, 0)
   158→   158→  const cashTotal = delivered.filter((c) => c.payment === 'cash').reduce((s, c) => s + c.price, 0)
   159→   159→  const cardTotal = delivered.filter((c) => c.payment === 'card').reduce((s, c) => s + c.price, 0)
   160→   160→
   161→   161→  return (
   162→   162→    <div className="grid grid-cols-1 lg:grid-cols-[1fr_360px] gap-6">
   163→   163→      {/* Pizarra */}
   164→   164→      <div>
   165→   165→        <BookmarkTabs
   166→   166→          active={activeTag}
   167→   167→          onChange={setActiveTag}
   168→   168→          tabs={tags.map((t) => ({ id: t, label: t === 'todos' ? 'Todos' : t }))}
   169→   169→        />
   170→   170→        <div className="relative mb-4">
   171→   171→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   172→   172→          <input
   173→   173→            type="search"
   174→   174→            value={search}
   175→   175→            onChange={(e) => setSearch(e.target.value)}
   176→   176→            placeholder="Buscar producto…"
   177→   177→            className="input-wellness pl-9 pr-9"
   178→   178→          />
   179→   179→          {search && (
   180→   180→            <button
   181→   181→              onClick={() => setSearch('')}
   182→   182→              className="absolute right-2 top-1/2 -translate-y-1/2 p-1 text-muted-foreground hover:text-foreground"
   183→   183→              aria-label="Limpiar"
   184→   184→            >
   185→   185→              <X className="w-3.5 h-3.5" />
   186→   186→            </button>
   187→   187→          )}
   188→   188→        </div>
   189→   189→
   190→   190→        {filtered.length === 0 ? (
   191→   191→          <EmptyState icon={<PackageIcon className="w-6 h-6" />} title="Sin productos" />
   192→   192→        ) : (
   193→   193→          <DndContext sensors={sensors} collisionDetection={closestCenter} onDragEnd={onDragEnd}>
   194→   194→            <SortableContext
   195→   195→              items={search || activeTag !== 'todos' ? [] : filtered.map((p) => p.id)}
   196→   196→              strategy={verticalListSortingStrategy}
   197→   197→            >
   198→   198→              <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-3 xl:grid-cols-4 gap-3">
   199→   199→                {filtered.map((p) => {
   200→   200→                  const productProtocol = protocols.find((pr) => pr.productId === p.id)
   201→   201→                  return (
   202→   202→                    <ProductCard
   203→   203→                      key={p.id}
   204→   204→                      product={p}
   205→   205→                      draggable={!search && activeTag === 'todos'}
   206→   206→                      hasProtocol={!!productProtocol}
   207→   207→                      onProtocol={() => productProtocol && setProtocolModal(productProtocol)}
   208→   208→                      onClick={() => setPaymentModal(p)}
   209→   209→                    />
   210→   210→                  )
   211→   211→                })}
   212→   212→              </div>
   213→   213→            </SortableContext>
   214→   214→          </DndContext>
   215→   215→        )}
   216→   216→      </div>
   217→   217→
   218→   218→      {/* Comandas panel */}
   219→   219→      <div className="lg:sticky lg:top-20 self-start">
   220→   220→        <Card>
   221→   221→          <div className="flex items-center justify-between mb-3">
   222→   222→            <h3 className="font-serif text-lg">Comandas simuladas</h3>
   223→   223→            <Badge variant="muted">{pending.length} pendientes</Badge>
   224→   224→          </div>
   225→   225→
   226→   226→          {comandas.length === 0 ? (
   227→   227→            <p className="text-sm text-muted-foreground py-6 text-center">
   228→   228→              Haz click en un producto para crear una venta simulada.
   229→   229→            </p>
   230→   230→          ) : (
   231→   231→            <ul className="space-y-2 max-h-[420px] overflow-y-auto custom-scroll pr-1">
   232→   232→              {pending
   233→   233→                .sort((a, b) => a.priority - b.priority)
   234→   234→                .map((c) => (
   235→   235→                  <ComandaItem
   236→   236→                    key={c.id}
   237→   237→                    comanda={c}
   238→   238→                    onUp={() => move(c.id, -1)}
   239→   239→                    onDown={() => move(c.id, 1)}
   240→   240→                    onDone={() => markDone(c.id)}
   241→   241→                    onRemove={() => remove(c.id)}
   242→   242→                  />
   243→   243→                ))}
   244→   244→              {delivered.map((c) => (
   245→   245→                <li key={c.id} className="flex items-center justify-between p-2 rounded-lg bg-muted/30 opacity-60">
   246→   246→                  <span className="text-sm line-through">{c.name}</span>
   247→   247→                  <span className="text-sm text-sage">{eur(c.price)}</span>
   248→   248→                </li>
   249→   249→              ))}
   250→   250→            </ul>
   251→   251→          )}
   252→   252→
   253→   253→          <div className="mt-4 pt-4 border-t border-border space-y-1.5 text-sm">
   254→   254→            <Row label="Total vendido" value={eur(total)} bold />
   255→   255→            <Row label="Efectivo" value={eur(cashTotal)} icon={<Banknote className="w-3.5 h-3.5" />} />
   256→   256→            <Row label="Tarjeta" value={eur(cardTotal)} icon={<CreditCard className="w-3.5 h-3.5" />} />
   257→   257→          </div>
   258→   258→        </Card>
   259→   259→      </div>
   260→   260→
   261→   261→      {paymentModal && (
   262→   262→        <PaymentModal
   263→   263→          product={paymentModal}
   264→   264→          onClose={() => setPaymentModal(null)}
   265→   265→          onSelect={(m) => addSale(paymentModal, m)}
   266→   266→        />
   267→   267→      )}
   268→   268→
   269→   269→      {protocolModal && (
   270→   270→        <ProtocolModal protocol={protocolModal} onClose={() => setProtocolModal(null)} />
   271→   271→      )}
   272→   272→    </div>
   273→   273→  )
   274→   274→}

   278→   278→function ProductCard({
   279→   279→  product,
   280→   280→  draggable,
   281→   281→  hasProtocol,
   282→   282→  onProtocol,
   283→   283→  onClick,
   284→   284→}: {
   285→   285→  product: Product
   286→   286→  draggable: boolean
   287→   287→  hasProtocol: boolean
   288→   288→  onProtocol: () => void
   289→   289→  onClick: () => void
   290→   290→}) {
   291→   291→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({
   292→   292→    id: product.id,
   293→   293→    disabled: !draggable,
   294→   294→  })
   295→   295→  const style: React.CSSProperties = {
   296→   296→    transform: CSS.Transform.toString(transform),
   297→   297→    transition,
   298→   298→    opacity: isDragging ? 0.5 : 1,
   299→   299→  }
   300→   300→  return (
   301→   301→    <div
   302→   302→      ref={setNodeRef}
   303→   303→      style={style}
   304→   304→      {...attributes}
   305→   305→      className={`card-wellness p-3 flex flex-col gap-2 ${draggable ? 'cursor-grab active:cursor-grabbing' : 'cursor-pointer'}`}
   306→   306→    >
   307→   307→      <div
   308→   308→        {...listeners}
   309→   309→        className="aspect-square rounded-lg bg-muted flex items-center justify-center overflow-hidden"
   310→   310→        onClick={onClick}
   311→   311→      >
   312→   312→        {product.imageUrl ? (
   313→   313→          <img src={product.imageUrl} alt={product.name} className="w-full h-full object-cover" />
   314→   314→        ) : (
   315→   315→          <PackageIcon className="w-8 h-8 text-muted-foreground" />
   316→   316→        )}
   317→   317→      </div>
   318→   318→      <div className="flex items-start justify-between gap-1">
   319→   319→        <p className="text-sm font-medium leading-tight">{product.name}</p>
   320→   320→        {hasProtocol && (
   321→   321→          <button
   322→   322→            className="btn-ghost p-1 -mt-1 -mr-1 text-sage"
   323→   323→            onClick={(e) => {
   324→   324→              e.stopPropagation()
   325→   325→              onProtocol()
   326→   326→            }}
   327→   327→            aria-label="Ver protocolo"
   328→   328→            title="Protocolo"
   329→   329→          >
   330→   330→            <ScrollText className="w-3.5 h-3.5" />
   331→   331→          </button>
   332→   332→        )}
   333→   333→      </div>
   334→   334→      <p className="text-sage font-semibold">{eur(product.price)}</p>
   335→   335→    </div>
   336→   336→  )
   337→   337→}

   341→   341→  onUp,

   343→   343→  onDone,

   429→   429→function ProtocolModal({ protocol, onClose }: { protocol: Protocol; onClose: () => void }) {

   452→   452→function Row({
   453→   453→  label,
   454→   454→  value,
   455→   455→  icon,
   456→   456→  bold,
   457→   457→}: {
   458→   458→  label: string

   480→   480→export function LivePizarra({ employeeId }: { employeeId: string }) {
   481→   481→  const [products, setProducts] = useState<Product[]>([])
   482→   482→  const [protocols, setProtocols] = useState<Protocol[]>([])
   483→   483→  const [order, setOrder] = useState<string[]>([])
   484→   484→  const [search, setSearch] = useState('')
   485→   485→  const [activeTag, setActiveTag] = useState('todos')
   486→   486→  const [paymentModal, setPaymentModal] = useState<Product | null>(null)
   487→   487→  const [protocolModal, setProtocolModal] = useState<Protocol | null>(null)
   488→   488→  const [comandas, setComandas] = useState<LiveComanda[]>([])
   489→   489→  const [loading, setLoading] = useState(true)
   490→   490→  const refreshTimer = useRef<number | null>(null)
   491→   491→
   492→   492→  useEffect(() => {
   493→   493→    ;(async () => {
   494→   494→      try {
   495→   495→        const [p, prots] = await Promise.all([
   496→   496→          get<Product[]>('/api/products?active=true'),
   497→   497→          get<Protocol[]>('/api/protocols'),
   498→   498→        ])
   499→   499→        setProducts(p)
   500→   500→        setProtocols(prots)
   501→   501→        setOrder(p.map((x) => x.id))
   502→   502→      } catch {
   503→   503→        toast.error('No se pudieron cargar productos')
   504→   504→      } finally {
   505→   505→        setLoading(false)
   506→   506→      }
   507→   507→    })()
   508→   508→  }, [])
   509→   509→
   510→   510→  // Refresh pending sale items every 5s and on focus
   511→   511→  async function loadComandas() {
   512→   512→    try {
   513→   513→      // We need all pending sale items for the employee — fetch recent sales and pull items
   514→   514→      const sales = await get<any[]>(`/api/sales?employeeId=${employeeId}`)
   515→   515→      const items: LiveComanda[] = []
   516→   516→      for (const s of sales) {
   517→   517→        for (const it of s.items || []) {
   518→   518→          if (it.status === 'PENDING') {
   519→   519→            items.push({ ...it, payment: s.paymentMethod, name: it.productName })
   520→   520→          }
   521→   521→        }
   522→   522→      }
   523→   523→      // Also load any DELIVERED in the last 24h for totals
   524→   524→      const now = new Date()
   525→   525→      const cutoff = new Date(now.getTime() - 24 * 60 * 60 * 1000)
   526→   526→      const deliveredToday: LiveComanda[] = []
   527→   527→      for (const s of sales) {
   528→   528→        if (new Date(s.createdAt) < cutoff) continue
   529→   529→        for (const it of s.items || []) {
   530→   530→          if (it.status === 'DELIVERED') {
   531→   531→            deliveredToday.push({ ...it, payment: s.paymentMethod, name: it.productName })
   532→   532→          }
   533→   533→        }
   534→   534→      }
   535→   535→      setComandas([...items.sort((a, b) => a.priority - b.priority), ...deliveredToday])
   536→   536→    } catch {
   537→   537→      // ignore
   538→   538→    }
   539→   539→  }
   540→   540→
   541→   541→  useEffect(() => {
   542→   542→    loadComandas()
   543→   543→    refreshTimer.current = window.setInterval(loadComandas, 8000)
   544→   544→    return () => {
   545→   545→      if (refreshTimer.current) window.clearInterval(refreshTimer.current)
   546→   546→    }
   547→   547→  }, [employeeId])
   548→   548→
   549→   549→  const tags = useMemo(() => {
   550→   550→    const s = new Set<string>()
   551→   551→    for (const p of products) for (const t of p.tags || []) s.add(t)
   552→   552→    return ['todos', ...Array.from(s).sort()]
   553→   553→  }, [products])
   554→   554→
   555→   555→  const filtered = useMemo(() => {
   556→   556→    const q = search.trim().toLowerCase()
   557→   557→    return order
   558→   558→      .map((id) => products.find((p) => p.id === id))
   559→   559→      .filter(Boolean)
   560→   560→      .filter((p) => (activeTag === 'todos' ? true : (p!.tags || []).includes(activeTag)))
   561→   561→      .filter((p) => (q ? p!.name.toLowerCase().includes(q) : true)) as Product[]
   562→   562→  }, [order, products, search, activeTag])
   563→   563→
   564→   564→  const sensors = useSensors(
   565→   565→    useSensor(PointerSensor, { activationConstraint: { distance: 6 } }),
   566→   566→    useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates })
   567→   567→  )
   568→   568→  function onDragEnd(e: DragEndEvent) {
   569→   569→    const { active, over } = e
   570→   570→    if (!over || active.id === over.id) return
   571→   571→    setOrder((prev) => {
   572→   572→      const oldI = prev.indexOf(active.id as string)
   573→   573→      const newI = prev.indexOf(over.id as string)
   574→   574→      return arrayMove(prev, oldI, newI)
   575→   575→    })
   576→   576→  }

   578→   578→  async function addSale(product: Product, payment: PaymentMethod) {
   579→   579→    try {
   580→   580→      await post('/api/sales', {
   581→   581→        employeeId,
   582→   582→        items: [{ productId: product.id, productName: product.name, price: product.price, quantity: 1 }],
   583→   583→        paymentMethod: payment,
   584→   584→        total: product.price,
   585→   585→      })
   586→   586→      toast.success(`Venta registrada: ${product.name}`)
   587→   587→      setPaymentModal(null)
   588→   588→      loadComandas()
   589→   589→    } catch {
   590→   590→      toast.error('No se pudo registrar la venta')
   591→   591→    }
   592→   592→  }

   597→   597→      toast.success('Comanda entregada')

   604→   604→  async function removeItem(itemId: string) {

   606→   606→    try {
   607→   607→      await del(`/api/sale-items/${itemId}`)
   608→   608→      toast.success('Comanda eliminada')
   609→   609→      loadComandas()
   610→   610→    } catch {
   611→   611→      toast.error('No se pudo eliminar')
   612→   612→    }

   615→   615→  async function movePriority(itemId: string, dir: -1 | 1) {
   616→   616→    const pending = comandas.filter((c) => c.status === 'PENDING').sort((a, b) => a.priority - b.priority)
   617→   617→    const idx = pending.findIndex((c) => c.id === itemId)
   618→   618→    if (idx === -1) return
   619→   619→    const newIdx = idx + dir
   620→   620→    if (newIdx < 0 || newIdx >= pending.length) return
   621→   621→    const a = pending[idx]
   622→   622→    const b = pending[newIdx]
   623→   623→    // swap priorities via two PUTs
   624→   624→    try {
   625→   625→      await Promise.all([
   626→   626→        put(`/api/sale-items/${a.id}`, { priority: b.priority }),
   627→   627→        put(`/api/sale-items/${b.id}`, { priority: a.priority }),
   628→   628→      ])
   629→   629→      loadComandas()
   630→   630→    } catch {
   631→   631→      toast.error('No se pudo reordenar')
   632→   632→    }
   633→   633→  }
   634→   634→
   635→   635→  const pending = comandas.filter((c) => c.status === 'PENDING')
   636→   636→  const delivered = comandas.filter((c) => c.status === 'DELIVERED')
   637→   637→  const total = delivered.reduce((s, c) => s + c.price * c.quantity, 0)
   638→   638→  const cashTotal = delivered.filter((c) => c.payment === 'cash').reduce((s, c) => s + c.price * c.quantity, 0)
   639→   639→  const cardTotal = delivered.filter((c) => c.payment === 'card').reduce((s, c) => s + c.price * c.quantity, 0)
   640→   640→
   641→   641→  return (
   642→   642→    <div className="grid grid-cols-1 lg:grid-cols-[1fr_360px] gap-6">
   643→   643→      {/* Pizarra */}
   644→   644→      <div>
   645→   645→        <BookmarkTabs
   646→   646→          active={activeTag}
   647→   647→          onChange={setActiveTag}
   648→   648→          tabs={tags.map((t) => ({ id: t, label: t === 'todos' ? 'Todos' : t }))}
   649→   649→        />
   650→   650→        <div className="relative mb-4">
   651→   651→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   652→   652→          <input
   653→   653→            type="search"
   654→   654→            value={search}
   655→   655→            onChange={(e) => setSearch(e.target.value)}
   656→   656→            placeholder="Buscar producto…"
   657→   657→            className="input-wellness pl-9 pr-9"
   658→   658→          />
   659→   659→          {search && (
   660→   660→            <button
   661→   661→              onClick={() => setSearch('')}
   662→   662→              className="absolute right-2 top-1/2 -translate-y-1/2 p-1 text-muted-foreground hover:text-foreground"
   663→   663→              aria-label="Limpiar"
   664→   664→            >
   665→   665→              <X className="w-3.5 h-3.5" />
   666→   666→            </button>
   667→   667→          )}
   668→   668→        </div>
   669→   669→
   670→   670→        {loading ? (
   671→   671→          <LoadingBlock label="Cargando productos…" />
   672→   672→        ) : filtered.length === 0 ? (
   673→   673→          <EmptyState icon={<PackageIcon className="w-6 h-6" />} title="Sin productos" />
   674→   674→        ) : (
   675→   675→          <DndContext sensors={sensors} collisionDetection={closestCenter} onDragEnd={onDragEnd}>
   676→   676→            <SortableContext
   677→   677→              items={!search && activeTag === 'todos' ? filtered.map((p) => p.id) : []}
   678→   678→              strategy={verticalListSortingStrategy}
   679→   679→            >
   680→   680→              <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-3 xl:grid-cols-4 gap-3">
   681→   681→                {filtered.map((p) => {
   682→   682→                  const productProtocol = protocols.find((pr) => pr.productId === p.id)
   683→   683→                  return (
   684→   684→                    <ProductCard
   685→   685→                      key={p.id}
   686→   686→                      product={p}
   687→   687→                      draggable={!search && activeTag === 'todos'}
   688→   688→                      hasProtocol={!!productProtocol}
   689→   689→                      onProtocol={() => productProtocol && setProtocolModal(productProtocol)}
   690→   690→                      onClick={() => setPaymentModal(p)}
   691→   691→                    />
   692→   692→                  )
   693→   693→                })}
   694→   694→              </div>
   695→   695→            </SortableContext>
   696→   696→          </DndContext>
   697→   697→        )}
   698→   698→      </div>
   699→   699→
   700→   700→      {/* Comandas panel */}
   701→   701→      <div className="lg:sticky lg:top-20 self-start">
   702→   702→        <Card>
   703→   703→          <div className="flex items-center justify-between mb-3">
   704→   704→            <h3 className="font-serif text-lg">Comandas</h3>
   705→   705→            <Badge variant="muted">{pending.length} pendientes</Badge>
   706→   706→          </div>
   707→   707→          {comandas.length === 0 ? (
   708→   708→            <p className="text-sm text-muted-foreground py-6 text-center">
   709→   709→              Haz click en un producto para registrar una venta.
   710→   710→            </p>
   711→   711→          ) : (
   712→   712→            <ul className="space-y-2 max-h-[420px] overflow-y-auto custom-scroll pr-1">
   713→   713→              {pending
   714→   714→                .sort((a, b) => a.priority - b.priority)
   715→   715→                .map((c) => (
   716→   716→                  <ComandaItem
   717→   717→                    key={c.id}
   718→   718→                    comanda={c}
   719→   719→                    onUp={() => movePriority(c.id, -1)}
   720→   720→                    onDown={() => movePriority(c.id, 1)}
   721→   721→                    onDone={() => markDone(c.id)}
   722→   722→                    onRemove={() => removeItem(c.id)}
   723→   723→                  />
   724→   724→                ))}
   725→   725→              {delivered.length > 0 && (
   726→   726→                <li className="pt-2 mt-2 border-t border-border">
   727→   727→                  <p className="text-xs uppercase tracking-wide text-muted-foreground mb-2">
   728→   728→                    Entregadas hoy ({delivered.length})
   729→   729→                  </p>
   730→   730→                  <div className="space-y-1">
   731→   731→                    {delivered.slice(0, 20).map((c) => (
   732→   732→                      <div key={c.id} className="flex items-center justify-between text-xs py-1 opacity-70">
   733→   733→                        <span className="truncate">{c.productName}</span>
   734→   734→                        <span className="text-sage">{eur(c.price * c.quantity)}</span>
   735→   735→                      </div>
   736→   736→                    ))}
   737→   737→                  </div>
   738→   738→                </li>
   739→   739→              )}
   740→   740→            </ul>
   741→   741→          )}
   742→   742→          <div className="mt-4 pt-4 border-t border-border space-y-1.5 text-sm">
   743→   743→            <Row label="Total vendido" value={eur(total)} bold />
   744→   744→            <Row label="Efectivo" value={eur(cashTotal)} icon={<Banknote className="w-3.5 h-3.5" />} />
   745→   745→            <Row label="Tarjeta" value={eur(cardTotal)} icon={<CreditCard className="w-3.5 h-3.5" />} />
   746→   746→          </div>
   747→   747→        </Card>
   748→   748→      </div>
   749→   749→
   750→   750→      {paymentModal && (
   751→   751→        <PaymentModal
   752→   752→          product={paymentModal}
   753→   753→          onClose={() => setPaymentModal(null)}
   754→   754→          onSelect={(m) => addSale(paymentModal, m)}
   755→   755→        />
   756→   756→      )}
   757→   757→      {protocolModal && (
   758→   758→        <ProtocolModal protocol={protocolModal} onClose={() => setProtocolModal(null)} />
   759→   759→      )}
   760→   760→    </div>
   761→   761→  )
   762→   762→}
